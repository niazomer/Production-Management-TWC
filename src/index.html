    async function loadState() {

        const stored = localStorage.getItem(STORAGE_KEY);

        if (stored) {

            appState = JSON.parse(stored);

            if(!appState.passwords) appState.passwords = { 'Manager': '' };

            if(!appState.profileImages) appState.profileImages = {};

            if(!appState.notifications) appState.notifications = {};

            if(!appState.transactions) appState.transactions = [];

            if(!appState.inbox) appState.inbox = [];

        }


        if (supabaseClient) {

            try {

                const { data: deliverablesData } = await supabaseClient.from('deliverables').select('*');

                if (deliverablesData && deliverablesData.length > 0) {

                    appState.deliverables = deliverablesData;

                }


                const { data: metaRows } = await supabaseClient.from('app_meta').select('*').eq('key', 'global');

                if (metaRows && metaRows.length > 0 && metaRows[0].data) {

                    const d = metaRows[0].data;

                    if(d.editors) appState.editors = d.editors;

                    if(d.rates) appState.rates = d.rates;

                    if(d.passwords) appState.passwords = d.passwords;

                    if(d.profileImages) appState.profileImages = d.profileImages;

                    if(d.notifications) appState.notifications = d.notifications;

                    if(d.transactions) appState.transactions = d.transactions;

                    if(d.inbox) appState.inbox = d.inbox;

                }

                if (currentUser) refreshCurrentView();

                console.warn("Supabase fetch fallback:", err);

            }

        }

    }


    async function saveState() {

        localStorage.setItem(STORAGE_KEY, JSON.stringify(appState));

        

        if (supabaseClient) {

            try {

                if (appState.deliverables.length > 0) {

                    await supabaseClient.from('deliverables').upsert(appState.deliverables);

                const metaPayload = {

                    editors: appState.editors,

                    rates: appState.rates,

                    passwords: appState.passwords,

                    profileImages: appState.profileImages,

                    transactions: appState.transactions,

                    inbox: appState.inbox

                await supabaseClient.from('app_meta').upsert({ key: 'global', data: metaPayload });

            } catch (err) {

            }

        }

    }

    async function submitPayment() {

        const editor = document.getElementById('payment-editor-id').value;

        const amount = Number(document.getElementById('payment-amount').value);

        if(amount <= 0) return;


        let remaining = amount;

        let tasks = appState.deliverables.filter(d => d.editor === editor && d.status === 'Completed');

        tasks.sort((a,b) => new Date(a.completed) - new Date(b.completed));


        tasks.forEach(t => {

            if(remaining <= 0) return;

            const taskTotal = t.overrideRate !== null ? Number(t.overrideRate) : appState.rates[t.type];

            const taskPending = taskTotal - (t.paidAmount || 0);


            if(taskPending > 0) {

                if(remaining >= taskPending) {

                    t.paidAmount = (t.paidAmount || 0) + taskPending;

                    t.paidDate = getTodayStr();

                    remaining -= taskPending;

                } else {

                    t.paidAmount = (t.paidAmount || 0) + remaining;

                    t.paidDate = getTodayStr();

                    remaining = 0;

                }

            }

        });


        appState.transactions.push({

            id: Date.now().toString(),

            date: getTodayStr(),

            time: new Date().toLocaleTimeString('en-US', { hour12: false }),

            editor: editor,

            amount: amount

        });


        if(!appState.notifications) appState.notifications = {};

        if(!appState.notifications[editor]) appState.notifications[editor] = [];

        appState.notifications[editor].push(`The studio administrator has processed a payment of ${formatINR(amount)} to your account.`);


        await saveState();

        closeModal('modal-payment');

        renderTotalPayouts();

        showToast("Payment recorded successfully");

    }

    async function handleSaveTask(e) {

        e.preventDefault();

        const id = document.getElementById('task-id').value;

        const overVal = document.getElementById('task-override').value;


        const data = {

            client: document.getElementById('task-client').value.trim(),

            type: document.getElementById('task-type').value,

            deadline: document.getElementById('task-deadline').value,

            editor: isAdmin ? document.getElementById('task-editor').value : currentUser,

            status: isAdmin ? document.getElementById('task-status').value : 'Not Started',

            overrideRate: (isAdmin && overVal) ? Number(overVal) : null,

        };


        if (id) {

            const idx = appState.deliverables.findIndex(d => d.id === id);

            if(data.status === 'Completed' && appState.deliverables[idx].status !== 'Completed') {

                data.completed = getTodayStr();

                data.paidAmount = 0;

            } else if (data.status !== 'Completed') {

                data.completed = null;

                data.paidAmount = 0;

            }

            appState.deliverables[idx] = { ...appState.deliverables[idx], ...data };

        } else {

            appState.deliverables.push({ id: Date.now().toString(), created: getTodayStr(), paidAmount: 0, ...data });

        }


        await saveState();

        closeModal('modal-task');

        refreshCurrentView();

        buildClientSuggestions();

        showToast("Deliverable saved");

    }
